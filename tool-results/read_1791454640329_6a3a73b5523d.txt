    86→    setLoading(true);
    87→    const params = new URLSearchParams();
    88→    Object.entries(filters).forEach(([k, v]) => {
    89→      if (v !== "all") params.set(k, v);
    90→    });
    91→    try {
    92→      const res = await fetch(`/api/kittens?${params.toString()}`);
    93→      const data = await res.json();
    94→      setKittens(data.kittens ?? []);
    95→    } catch {
    96→      setKittens([]);
    97→    } finally {
    98→      setLoading(false);
    99→    }
   100→  }, [filters]);
   101→
   102→  useEffect(() => {
   103→    const t = setTimeout(fetchKittens, 150);
   104→    return () => clearTimeout(t);
   105→  }, [fetchKittens]);
   106→
   107→  useEffect(() => {
   108→    setActiveCount(
   109→      Object.values(filters).filter((v) => v !== "all").length
   110→    );
   111→  }, [filters]);
   112→
   113→  // Listen for filter requests from the personality quiz

   143→            <PawPrint className="h-3.5 w-3.5 mr-1.5" />
   144→            Каталог котят · с фильтрами
   145→          </Badge>
   146→          <h2 className="font-serif text-3xl sm:text-4xl lg:text-5xl font-semibold text-foreground leading-tight mb-5">
   147→            Найдите котёнка под характер семьи
   148→          </h2>
   149→          <p className="text-lg text-muted-foreground leading-relaxed">
   150→            Фильтр по характеру — то, чего нет ни у одного конкурента. Выбирайте
   151→            не по окрасу, а по темпераменту: спокойный для детей, игривый для
   152→            подростков, ласковый для одинокого хозяина.
   153→          </p>
   154→          <div className="mt-5">
   155→            <LiveAvailability />
   156→          </div>
   157→        </div>
   158→
   159→        {/* Filters panel */}
   160→        <div className="rounded-2xl bg-muted/40 ring-1 ring-border p-4 sm:p-6 mb-8">
   161→          <div className="flex items-center justify-between mb-4">
   162→            <div className="flex items-center gap-2">
   163→              <Filter className="h-5 w-5 text-primary" />
   164→              <h3 className="font-serif text-lg font-semibold text-foreground">
   165→                Фильтры

   168→                <Badge className="bg-primary text-primary-foreground ml-1">
   169→                  {activeCount}
   170→                </Badge>
   171→              )}
   172→            </div>
   173→            {activeCount > 0 && (
   174→              <Button
   175→                variant="ghost"
   176→                size="sm"
   177→                onClick={resetFilters}
   178→                className="text-muted-foreground hover:text-foreground"
   179→              >
   180→                <X className="h-4 w-4 mr-1" />
   181→                Сбросить

   188→            <FilterGroup label="Окрас">

   196→                    {opt.label}

   206→                  <FilterChip
   207→                    key={opt.value}
   208→                    active={filters.gender === opt.value}
   209→                    onClick={() => updateFilter("gender", opt.value)}
   210→                  >
   211→                    {opt.label}

   213→                ))}

   227→                    <FilterChip
   228→                      key={opt.value}
   229→                      active={filters.personality === opt.value}
   230→                      onClick={() => updateFilter("personality", opt.value)}

   249→                <FilterChip
   250→                  active={filters.status === "available"}
   251→                  onClick={() => updateFilter("status", "available")}
   252→                >
   253→                  Доступны
   254→                </FilterChip>

   256→                  active={filters.status === "expected"}

   261→              </div>
   262→            </FilterGroup>
   263→          </div>
   264→
   265→          {/* Result count */}
   266→          <div className="mt-4 pt-4 border-t border-border flex items-center justify-between flex-wrap gap-2">
   267→            <p className="text-sm text-muted-foreground">
   268→              {loading ? (
   269→                "Загрузка..."
   270→              ) : (
   271→                <>
   272→                  Найдено: <span className="font-semibold text-foreground">{kittens.length}</span> котят

   274→                    <>, из них <span className="font-semibold text-primary">{availableCount}</span> доступно</>
   275→                  )}
   276→                </>
   277→              )}
   278→            </p>
   279→          </div>
   280→        </div>
   281→
   282→        {/* Kittens grid */}
   283→        {loading ? (
   284→          <div className="grid sm:grid-cols-2 lg:grid-cols-3 gap-5">
   285→            {[1, 2, 3, 4, 5, 6].map((i) => (
   286→              <Skeleton key={i} className="h-[420px] rounded-2xl" />
   287→            ))}
   288→          </div>
   289→        ) : kittens.length === 0 ? (
   290→          <div className="text-center py-16 rounded-2xl bg-muted/30 ring-1 ring-border">
   291→            <PawPrint className="h-12 w-12 text-muted-foreground/40 mx-auto mb-4" />
   292→            <h3 className="font-serif text-xl font-semibold text-foreground mb-2">
   293→              Котят по фильтрам не нашлось

   295→            <p className="text-muted-foreground mb-5 max-w-md mx-auto">
   296→              Попробуйте смягчить фильтры или запишитесь в лист ожидания — мы

   298→            </p>
   299→            <Button onClick={resetFilters} variant="outline">
   300→              Сбросить фильтры
   301→            </Button>
   302→          </div>
   303→        ) : (
   304→          <div
   305→            key={`${filters.color}-${filters.gender}-${filters.personality}-${filters.status}`}
   306→            className="grid sm:grid-cols-2 lg:grid-cols-3 gap-5"
   307→          >
   308→            {kittens.map((kitten, i) => (
   309→              <div
   310→                key={kitten.id}

   312→                style={{ animationDelay: `${Math.min(i, 8) * 60}ms` }}

   314→                <KittenCard

   319→              </div>
   320→            ))}
   321→          </div>
   322→        )}
   323→      </div>
   324→
   325→      {/* Kitten detail dialog */}
   326→      <KittenDialog
   327→        kitten={selected}
   328→        onClose={() => setSelected(null)}
   329→        onBook={(k) => openBooking({ id: k.id, name: k.name })}
   330→      />
   331→    </section>
   332→  );
   333→}

   385→function KittenCard({
   386→  kitten,
   387→  onSelect,
   388→  onBook,
   389→}: {
   390→  kitten: Kitten;
   391→  onSelect: () => void;
   392→  onBook: () => void;
   393→}) {
   394→  const PersonalityIcon = PERSONALITY_ICON[kitten.personality];
   395→  const gender = GENDER_LABEL[kitten.gender];
   396→  const GenderIcon = gender.icon;
   397→  const isAvailable = kitten.status === "available";
   398→  const isExpected = kitten.status === "expected";
   399→  const { isFavorite, toggle: toggleFav } = useFavorites();
   400→  const { isComparing, toggle: toggleCompare, canAdd, max } = useCompare();
   401→
   402→  const fav = isFavorite(kitten.id);
   403→  const comparing = isComparing(kitten.id);
   404→
   405→  const handleFav = (e: React.MouseEvent) => {
   406→    e.stopPropagation();
   407→    toggleFav(kitten.id);
   408→    toast(fav ? "Убрано из избранного" : "Добавлено в избранное", {
   409→      description: fav ? undefined : `${kitten.name} теперь в вашем списке.`,
   410→    });
   411→  };
   412→
   413→  const handleCompare = (e: React.MouseEvent) => {
   414→    e.stopPropagation();
   415→    if (!comparing && !canAdd) {
   416→      toast.warning(`Можно сравнить максимум ${max} котят`, {
   417→        description: "Уберите одного, чтобы добавить другого.",
   418→      });
   419→      return;
   420→    }
   421→    toggleCompare(kitten.id);
   422→    toast(comparing ? "Убрано из сравнения" : "Добавлено к сравнению", {
   423→      description: comparing ? undefined : `${kitten.name} в списке для сравнения.`,
   424→    });
   425→  };
   426→
   427→  const handleShare = async (e: React.MouseEvent) => {
   428→    e.stopPropagation();
   429→    const url = `${window.location.origin}/#kittens`;
   430→    const shareData = {
   431→      title: `${kitten.name} — ${kitten.colorLabel} мейн-кун | Тихий Дом`,
   432→      text: `Посмотрите котёнка ${kitten.name} — ${kitten.personalityLabel.toLowerCase()}, ${kitten.colorLabel.toLowerCase()}. ${kitten.price.toLocaleString("ru-RU")} ₽`,
   433→      url,
   434→    };
   435→    try {
   436→      if (navigator.share) {
   437→        await navigator.share(shareData);
   438→      } else {
   439→        await navigator.clipboard.writeText(url);
   440→        toast.success("Ссылка скопирована", {
   441→          description: `Поделитесь котёнком ${kitten.name} с близкими.`,
   442→        });
   443→      }
   444→    } catch {
   445→      // user cancelled share — no toast
   446→    }
   447→  };
   448→
   449→  return (
   450→    <SpotlightCard className="rounded-2xl ring-1 ring-border h-full">
   451→    <Card className="group overflow-hidden border-0 bg-card hover:shadow-lg transition-all duration-300 flex flex-col h-full">
   452→      <div
   453→        className="relative aspect-[4/3] overflow-hidden cursor-pointer"
   454→        onClick={onSelect}
   455→      >
   456→        <Image
   457→          src={kitten.imageUrl}
   458→          alt={`${kitten.name} — ${kitten.colorLabel}`}
   459→          fill
   460→          sizes="(max-width: 640px) 100vw, (max-width: 1024px) 50vw, 33vw"
   461→          className="object-cover transition-transform duration-500 group-hover:scale-105"
   462→          placeholder="blur"
   463→          blurDataURL={BLUR_DATA_URLS.muted}
   464→        />
   465→        <div className="absolute inset-0 bg-gradient-to-t from-black/40 to-transparent" />
   466→
   467→        {/* Action buttons (top-right): favorite + compare */}
   468→        <div className="absolute top-2.5 right-2.5 flex flex-col gap-1.5 z-10">
   469→          <button
   470→            type="button"
   471→            onClick={handleFav}
   472→            aria-label={fav ? "Убрать из избранного" : "Добавить в избранное"}
   473→            aria-pressed={fav}
   474→            className="flex h-8 w-8 items-center justify-center rounded-full bg-background/90 backdrop-blur-sm border border-border text-foreground hover:bg-background hover:scale-110 transition-all shadow-sm"
   475→          >
   476→            <Heart
   477→              className={`h-4 w-4 transition-all ${
   478→                fav ? "fill-accent text-accent scale-110" : "text-foreground"
   479→              }`}
   480→            />
   481→          </button>
   482→          <button
   483→            type="button"
   484→            onClick={handleCompare}
   485→            aria-label={comparing ? "Убрать из сравнения" : "Добавить к сравнению"}
   486→            aria-pressed={comparing}
   487→            className={`flex h-8 w-8 items-center justify-center rounded-full backdrop-blur-sm border transition-all shadow-sm ${
   488→              comparing
   489→                ? "bg-secondary text-secondary-foreground border-secondary scale-110"
   490→                : "bg-background/90 border-border text-foreground hover:bg-background hover:scale-110"
   491→            }`}
   492→          >
   493→            <GitCompare className="h-4 w-4" />
   494→          </button>
   495→          <button
   496→            type="button"
   497→            onClick={handleShare}
   498→            aria-label="Поделиться"
   499→            className="flex h-8 w-8 items-center justify-center rounded-full bg-background/90 backdrop-blur-sm border border-border text-foreground hover:bg-background hover:scale-110 transition-all shadow-sm"
   500→          >
   501→            <Share2 className="h-4 w-4" />
   502→          </button>
   503→        </div>
   504→
   505→        {/* Personality badge (top-left) */}
   506→        <span className="absolute top-3 left-3 inline-flex items-center gap-1 px-2.5 py-1 rounded-full text-xs font-medium bg-background/90 text-foreground backdrop-blur-sm border border-border">
   507→          <PersonalityIcon className="h-3 w-3 text-accent" />
   508→          {kitten.personalityLabel}
   509→        </span>
   510→
   511→        {/* Name + price on image */}
   512→        <div className="absolute bottom-0 left-0 right-0 p-3 flex items-end justify-between">
   513→          <div className="text-background">
   514→            <h4 className="font-serif text-xl font-semibold leading-tight">{kitten.name}</h4>
   515→            <p className="text-xs text-background/80">{kitten.colorLabel}</p>
   516→          </div>
   517→          <div className="text-background text-right">
   518→            <p className="font-serif text-lg font-semibold leading-tight">
   519→              {kitten.price.toLocaleString("ru-RU")} ₽
   520→            </p>
   521→          </div>
   522→        </div>
   523→      </div>
   524→
   525→      <CardContent className="p-4 flex-1 flex flex-col gap-3">
   526→        {/* Meta row */}
   527→        <div className="flex items-center gap-3 flex-wrap text-xs text-muted-foreground">
   528→          <span className="inline-flex items-center gap-1">
   529→            <GenderIcon className="h-3.5 w-3.5" />
   530→            {gender.label}
   531→          </span>
   532→          <span className="inline-flex items-center gap-1">
   533→            <CalendarDays className="h-3.5 w-3.5" />
   534→            {formatAge(kitten.birthDate)}
   535→          </span>
   536→          {kitten.vaccinated && (
   537→            <span className="inline-flex items-center gap-1 text-primary">
   538→              <Syringe className="h-3.5 w-3.5" />
   539→              Привит
   540→            </span>
   541→          )}
   542→        </div>
   543→
   544→        <p className="text-sm text-muted-foreground line-clamp-2 leading-relaxed flex-1">
   545→          {kitten.description}
   546→        </p>
   547→
   548→        {/* Parents mini */}
   549→        {kitten.litter && (kitten.litter.father || kitten.litter.mother) && (
   550→          <div className="flex items-center gap-1.5 text-xs text-muted-foreground bg-muted/50 rounded-md px-2.5 py-1.5">

   810→function formatAge(iso: string): string {
   811→  try {
   812→    const birth = new Date(iso);
   813→    const now = new Date();
   814→    const months =
   815→      (now.getFullYear() - birth.getFullYear()) * 12 +
   816→      (now.getMonth() - birth.getMonth());
   817→    if (months <= 0) return "новорождённый";
   818→    if (months < 12) return `${months} мес.`;
   819→    const years = Math.floor(months / 12);
   820→    const remMonths = months % 12;
   821→    return remMonths === 0
   822→      ? `${years} ${years === 1 ? "год" : "года"}`
   823→      : `${years} г. ${remMonths} мес.`;
   824→  } catch {
   825→    return "—";
   826→  }
