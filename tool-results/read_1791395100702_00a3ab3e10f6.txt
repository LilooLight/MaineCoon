    67→export function KittensCatalog() {
    68→  const [kittens, setKittens] = useState<Kitten[]>([]);
    69→  const [loading, setLoading] = useState(true);
    70→  const [filters, setFilters] = useState({
    71→    color: "all",
    72→    gender: "all",
    73→    personality: "all",
    74→    status: "all",
    75→  });
    76→  const [activeCount, setActiveCount] = useState(0);
    77→  const [selected, setSelected] = useState<Kitten | null>(null);
    78→  const { openBooking } = useBooking();
    79→
    80→  const fetchKittens = useCallback(async () => {
    81→    setLoading(true);
    82→    const params = new URLSearchParams();
    83→    Object.entries(filters).forEach(([k, v]) => {
    84→      if (v !== "all") params.set(k, v);
    85→    });
    86→    try {
    87→      const res = await fetch(`/api/kittens?${params.toString()}`);
    88→      const data = await res.json();
    89→      setKittens(data.kittens ?? []);
    90→    } catch {
    91→      setKittens([]);
    92→    } finally {
    93→      setLoading(false);
    94→    }
    95→  }, [filters]);
    96→
    97→  useEffect(() => {
    98→    const t = setTimeout(fetchKittens, 150);
    99→    return () => clearTimeout(t);
   100→  }, [fetchKittens]);
   101→
   102→  useEffect(() => {
   103→    setActiveCount(
   104→      Object.values(filters).filter((v) => v !== "all").length
   105→    );
   106→  }, [filters]);
   107→
   108→  // Listen for filter requests from the personality quiz
   109→  useEffect(() => {
   110→    const handler = (e: Event) => {
   111→      const detail = (e as CustomEvent).detail as {
   112→        personality?: Personality;
   113→      };
   114→      if (detail?.personality) {
   115→        setFilters((f) => ({ ...f, personality: detail.personality! }));
   116→      }
   117→    };
   118→    window.addEventListener("cattery:setFilter", handler);
   119→    return () => window.removeEventListener("cattery:setFilter", handler);
   120→  }, []);
   121→
   122→  const updateFilter = (key: keyof typeof filters, value: string) => {
   123→    setFilters((f) => ({ ...f, [key]: value }));
   124→  };
   125→
   126→  const resetFilters = () => {
   127→    setFilters({ color: "all", gender: "all", personality: "all", status: "all" });
   128→  };
   129→
   130→  const availableCount = kittens.filter((k) => k.status === "available").length;
   131→
   132→  return (
   133→    <section id="kittens" className="py-16 sm:py-24 bg-background">
   134→      <div className="container mx-auto max-w-7xl px-4 sm:px-6 lg:px-8">
   135→        {/* Header */}
   136→        <div className="max-w-3xl mb-10 lg:mb-14">
   137→          <Badge variant="outline" className="mb-4 border-accent/30 bg-accent/5 text-accent">
   138→            <PawPrint className="h-3.5 w-3.5 mr-1.5" />
   139→            Каталог котят · с фильтрами
   140→          </Badge>
   141→          <h2 className="font-serif text-3xl sm:text-4xl lg:text-5xl font-semibold text-foreground leading-tight mb-5">
   142→            Найдите котёнка под характер семьи
   143→          </h2>
   144→          <p className="text-lg text-muted-foreground leading-relaxed">
   145→            Фильтр по характеру — то, чего нет ни у одного конкурента. Выбирайте
   146→            не по окрасу, а по темпераменту: спокойный для детей, игривый для
   147→            подростков, ласковый для одинокого хозяина.
   148→          </p>
   149→          <div className="mt-5">
   150→            <LiveAvailability />
   151→          </div>
   152→        </div>
   153→
   154→        {/* Filters panel */}
   155→        <div className="rounded-2xl bg-muted/40 ring-1 ring-border p-4 sm:p-6 mb-8">
   156→          <div className="flex items-center justify-between mb-4">
   157→            <div className="flex items-center gap-2">
   158→              <Filter className="h-5 w-5 text-primary" />
   159→              <h3 className="font-serif text-lg font-semibold text-foreground">
   160→                Фильтры
   161→              </h3>
   162→              {activeCount > 0 && (
   163→                <Badge className="bg-primary text-primary-foreground ml-1">
   164→                  {activeCount}
   165→                </Badge>
   166→              )}
   167→            </div>
   168→            {activeCount > 0 && (
   169→              <Button
   170→                variant="ghost"
   171→                size="sm"
   172→                onClick={resetFilters}
   173→                className="text-muted-foreground hover:text-foreground"
   174→              >
   175→                <X className="h-4 w-4 mr-1" />
   176→                Сбросить
   177→              </Button>
   178→            )}
   179→          </div>
   180→
   181→          <div className="grid sm:grid-cols-2 lg:grid-cols-4 gap-4">
   182→            {/* Color filter */}
   183→            <FilterGroup label="Окрас">
   184→              <div className="flex flex-wrap gap-1.5">
   185→                {COLOR_OPTIONS.map((opt) => (
   186→                  <FilterChip
   187→                    key={opt.value}
   188→                    active={filters.color === opt.value}
   189→                    onClick={() => updateFilter("color", opt.value)}
   190→                  >
   191→                    {opt.label}
   192→                  </FilterChip>
   193→                ))}
   194→              </div>
   195→            </FilterGroup>
   196→
   197→            {/* Gender filter */}
   198→            <FilterGroup label="Пол">
   199→              <div className="flex flex-wrap gap-1.5">
   200→                {GENDER_OPTIONS.map((opt) => (
   201→                  <FilterChip
   202→                    key={opt.value}
   203→                    active={filters.gender === opt.value}
   204→                    onClick={() => updateFilter("gender", opt.value)}
   205→                  >
   206→                    {opt.label}
   207→                  </FilterChip>
   208→                ))}
   209→              </div>
   210→            </FilterGroup>
   211→
   212→            {/* Personality filter — the key differentiator */}
   213→            <FilterGroup label="Характер" highlight>
   214→              <div className="flex flex-wrap gap-1.5">
   215→                {PERSONALITY_OPTIONS.map((opt) => {
   216→                  const Icon =
   217→                    opt.icon === "Moon" ? Moon :
   218→                    opt.icon === "Sparkles" ? Sparkles :
   219→                    opt.icon === "Mountain" ? Mountain :
   220→                    opt.icon === "Heart" ? Heart : PawPrint;
   221→                  return (
   222→                    <FilterChip
   223→                      key={opt.value}
   224→                      active={filters.personality === opt.value}
   225→                      onClick={() => updateFilter("personality", opt.value)}
   226→                    >
   227→                      <Icon className="h-3 w-3 mr-1" />
   228→                      {opt.label}
   229→                    </FilterChip>
   230→                  );
   231→                })}
   232→              </div>
   233→            </FilterGroup>
   234→
   235→            {/* Status filter */}
   236→            <FilterGroup label="Статус">
   237→              <div className="flex flex-wrap gap-1.5">
   238→                <FilterChip
   239→                  active={filters.status === "all"}
   240→                  onClick={() => updateFilter("status", "all")}
   241→                >
   242→                  Все
   243→                </FilterChip>
   244→                <FilterChip
   245→                  active={filters.status === "available"}
   246→                  onClick={() => updateFilter("status", "available")}
   247→                >
   248→                  Доступны
   249→                </FilterChip>
   250→                <FilterChip
   251→                  active={filters.status === "expected"}
   252→                  onClick={() => updateFilter("status", "expected")}
   253→                >
   254→                  Ожидаются
   255→                </FilterChip>
   256→              </div>
   257→            </FilterGroup>
   258→          </div>
   259→
   260→          {/* Result count */}
   261→          <div className="mt-4 pt-4 border-t border-border flex items-center justify-between flex-wrap gap-2">
   262→            <p className="text-sm text-muted-foreground">
   263→              {loading ? (
   264→                "Загрузка..."
   265→              ) : (
   266→                <>
   267→                  Найдено: <span className="font-semibold text-foreground">{kittens.length}</span> котят
   268→                  {availableCount > 0 && (
   269→                    <>, из них <span className="font-semibold text-primary">{availableCount}</span> доступно</>
   270→                  )}
   271→                </>
   272→              )}
   273→            </p>
   274→          </div>
   275→        </div>
   276→
   277→        {/* Kittens grid */}
   278→        {loading ? (
   279→          <div className="grid sm:grid-cols-2 lg:grid-cols-3 gap-5">
   280→            {[1, 2, 3, 4, 5, 6].map((i) => (
   281→              <Skeleton key={i} className="h-[420px] rounded-2xl" />
   282→            ))}
   283→          </div>
   284→        ) : kittens.length === 0 ? (
   285→          <div className="text-center py-16 rounded-2xl bg-muted/30 ring-1 ring-border">
   286→            <PawPrint className="h-12 w-12 text-muted-foreground/40 mx-auto mb-4" />
   287→            <h3 className="font-serif text-xl font-semibold text-foreground mb-2">
   288→              Котят по фильтрам не нашлось
   289→            </h3>
   290→            <p className="text-muted-foreground mb-5 max-w-md mx-auto">
   291→              Попробуйте смягчить фильтры или запишитесь в лист ожидания — мы
   292→              сообщим, когда появится подходящий котёнок.
   293→            </p>
   294→            <Button onClick={resetFilters} variant="outline">
   295→              Сбросить фильтры
   296→            </Button>
   297→          </div>
   298→        ) : (
   299→          <div
   300→            key={`${filters.color}-${filters.gender}-${filters.personality}-${filters.status}`}
   301→            className="grid sm:grid-cols-2 lg:grid-cols-3 gap-5"
   302→          >
   303→            {kittens.map((kitten, i) => (
   304→              <div
   305→                key={kitten.id}
   306→                className="kitten-card-enter"
   307→                style={{ animationDelay: `${Math.min(i, 8) * 60}ms` }}
   308→              >
   309→                <KittenCard
   310→                  kitten={kitten}
   311→                  onSelect={() => setSelected(kitten)}
   312→                  onBook={() => openBooking({ id: kitten.id, name: kitten.name })}
   313→                />
   314→              </div>
   315→            ))}
   316→          </div>
   317→        )}
   318→      </div>
   319→
   320→      {/* Kitten detail dialog */}
   321→      <KittenDialog
   322→        kitten={selected}
   323→        onClose={() => setSelected(null)}
   324→        onBook={(k) => openBooking({ id: k.id, name: k.name })}
   325→      />
   326→    </section>
   327→  );
   328→}

   380→function KittenCard({
   381→  kitten,
   382→  onSelect,
   383→  onBook,
   384→}: {
   385→  kitten: Kitten;
   386→  onSelect: () => void;
   387→  onBook: () => void;
   388→}) {
   389→  const PersonalityIcon = PERSONALITY_ICON[kitten.personality];
   390→  const gender = GENDER_LABEL[kitten.gender];
   391→  const GenderIcon = gender.icon;
   392→  const isAvailable = kitten.status === "available";
   393→  const isExpected = kitten.status === "expected";
   394→  const { isFavorite, toggle: toggleFav } = useFavorites();
   395→  const { isComparing, toggle: toggleCompare, canAdd, max } = useCompare();
   396→
   397→  const fav = isFavorite(kitten.id);
   398→  const comparing = isComparing(kitten.id);
   399→
   400→  const handleFav = (e: React.MouseEvent) => {
   401→    e.stopPropagation();
   402→    toggleFav(kitten.id);
   403→    toast(fav ? "Убрано из избранного" : "Добавлено в избранное", {
   404→      description: fav ? undefined : `${kitten.name} теперь в вашем списке.`,
   405→    });
   406→  };
   407→
   408→  const handleCompare = (e: React.MouseEvent) => {
   409→    e.stopPropagation();
   410→    if (!comparing && !canAdd) {
   411→      toast.warning(`Можно сравнить максимум ${max} котят`, {
   412→        description: "Уберите одного, чтобы добавить другого.",
   413→      });
   414→      return;
   415→    }
   416→    toggleCompare(kitten.id);
   417→    toast(comparing ? "Убрано из сравнения" : "Добавлено к сравнению", {
   418→      description: comparing ? undefined : `${kitten.name} в списке для сравнения.`,
   419→    });
   420→  };
   421→
   422→  return (
   423→    <SpotlightCard className="rounded-2xl ring-1 ring-border h-full">
   424→    <Card className="group overflow-hidden border-0 bg-card hover:shadow-lg transition-all duration-300 flex flex-col h-full">
   425→      <div
   426→        className="relative aspect-[4/3] overflow-hidden cursor-pointer"
   427→        onClick={onSelect}
   428→      >
   429→        <Image
   430→          src={kitten.imageUrl}
   431→          alt={`${kitten.name} — ${kitten.colorLabel}`}
   432→          fill
   433→          sizes="(max-width: 640px) 100vw, (max-width: 1024px) 50vw, 33vw"
   434→          className="object-cover transition-transform duration-500 group-hover:scale-105"
   435→        />
   436→        <div className="absolute inset-0 bg-gradient-to-t from-black/40 to-transparent" />
   437→
   438→        {/* Action buttons (top-right): favorite + compare */}
   439→        <div className="absolute top-2.5 right-2.5 flex flex-col gap-1.5 z-10">
   440→          <button
   441→            type="button"
   442→            onClick={handleFav}
   443→            aria-label={fav ? "Убрать из избранного" : "Добавить в избранное"}
   444→            aria-pressed={fav}
   445→            className="flex h-8 w-8 items-center justify-center rounded-full bg-background/90 backdrop-blur-sm border border-border text-foreground hover:bg-background hover:scale-110 transition-all shadow-sm"
   446→          >
   447→            <Heart
   448→              className={`h-4 w-4 transition-all ${
   449→                fav ? "fill-accent text-accent scale-110" : "text-foreground"
   450→              }`}
   451→            />
   452→          </button>
   453→          <button
   454→            type="button"
   455→            onClick={handleCompare}
   456→            aria-label={comparing ? "Убрать из сравнения" : "Добавить к сравнению"}
   457→            aria-pressed={comparing}
   458→            className={`flex h-8 w-8 items-center justify-center rounded-full backdrop-blur-sm border transition-all shadow-sm ${
   459→              comparing
   460→                ? "bg-secondary text-secondary-foreground border-secondary scale-110"
   461→                : "bg-background/90 border-border text-foreground hover:bg-background hover:scale-110"
   462→            }`}
   463→          >
   464→            <GitCompare className="h-4 w-4" />
   465→          </button>
   466→        </div>
   467→
   468→        {/* Personality badge (top-left) */}
   469→        <span className="absolute top-3 left-3 inline-flex items-center gap-1 px-2.5 py-1 rounded-full text-xs font-medium bg-background/90 text-foreground backdrop-blur-sm border border-border">
   470→          <PersonalityIcon className="h-3 w-3 text-accent" />
   471→          {kitten.personalityLabel}
   472→        </span>
   473→
   474→        {/* Name + price on image */}
   475→        <div className="absolute bottom-0 left-0 right-0 p-3 flex items-end justify-between">
   476→          <div className="text-background">
   477→            <h4 className="font-serif text-xl font-semibold leading-tight">{kitten.name}</h4>
   478→            <p className="text-xs text-background/80">{kitten.colorLabel}</p>
   479→          </div>
   480→          <div className="text-background text-right">
   481→            <p className="font-serif text-lg font-semibold leading-tight">
   482→              {kitten.price.toLocaleString("ru-RU")} ₽
   483→            </p>
   484→          </div>
   485→        </div>
   486→      </div>
   487→
   488→      <CardContent className="p-4 flex-1 flex flex-col gap-3">
   489→        {/* Meta row */}
   490→        <div className="flex items-center gap-3 flex-wrap text-xs text-muted-foreground">
   491→          <span className="inline-flex items-center gap-1">
   492→            <GenderIcon className="h-3.5 w-3.5" />
   493→            {gender.label}
   494→          </span>
   495→          <span className="inline-flex items-center gap-1">
   496→            <CalendarDays className="h-3.5 w-3.5" />
   497→            {formatAge(kitten.birthDate)}
   498→          </span>
   499→          {kitten.vaccinated && (
   500→            <span className="inline-flex items-center gap-1 text-primary">
   501→              <Syringe className="h-3.5 w-3.5" />
   502→              Привит
   503→            </span>
   504→          )}
   505→        </div>
   506→
   507→        <p className="text-sm text-muted-foreground line-clamp-2 leading-relaxed flex-1">
   508→          {kitten.description}
   509→        </p>
   510→
   511→        {/* Parents mini */}
   512→        {kitten.litter && (kitten.litter.father || kitten.litter.mother) && (
   513→          <div className="flex items-center gap-1.5 text-xs text-muted-foreground bg-muted/50 rounded-md px-2.5 py-1.5">
   514→            <Users className="h-3.5 w-3.5 shrink-0" />
   515→            <span className="truncate">
   516→              Родители: {kitten.litter.father?.name ?? "—"} × {kitten.litter.mother?.name ?? "—"}
   517→            </span>
   518→          </div>
   519→        )}
   520→      </CardContent>
   521→
   522→      <CardFooter className="p-4 pt-0 gap-2">
   523→        <Button
   524→          variant="ghost"
   525→          size="sm"
   526→          onClick={onSelect}
   527→          className="flex-1 text-muted-foreground hover:text-foreground"
   528→        >
   529→          Подробнее
   530→          <ChevronRight className="h-4 w-4 ml-1" />
   531→        </Button>
   532→        <Button
   533→          size="sm"
   534→          onClick={onBook}
   535→          disabled={kitten.status === "adopted"}
   536→          className="flex-1 bg-accent text-accent-foreground hover:bg-accent/90 disabled:opacity-50"
   537→        >
   538→          {isExpected ? "Лист ожидания" : isAvailable ? "Забронировать" : "Узнать"}
   539→        </Button>
   540→      </CardFooter>
   541→    </Card>
   542→    </SpotlightCard>
   543→  );
   544→}

   546→function KittenDialog({
   547→  kitten,
   548→  onClose,
   549→  onBook,
   550→}: {
   551→  kitten: Kitten | null;
   552→  onClose: () => void;
   553→  onBook: (k: Kitten) => void;
   554→}) {
   555→  return (
   556→    <Dialog open={!!kitten} onOpenChange={(o) => !o && onClose()}>
   557→      <DialogContent className="max-w-3xl max-h-[90vh] overflow-y-auto custom-scroll bg-background p-0">

   766→function formatAge(iso: string): string {
   767→  try {
   768→    const birth = new Date(iso);
   769→    const now = new Date();
   770→    const months =
   771→      (now.getFullYear() - birth.getFullYear()) * 12 +
   772→      (now.getMonth() - birth.getMonth());
   773→    if (months <= 0) return "новорождённый";
   774→    if (months < 12) return `${months} мес.`;
   775→    const years = Math.floor(months / 12);
   776→    const remMonths = months % 12;
   777→    return remMonths === 0
   778→      ? `${years} ${years === 1 ? "год" : "года"}`
   779→      : `${years} г. ${remMonths} мес.`;
   780→  } catch {
   781→    return "—";
   782→  }
   783→}
