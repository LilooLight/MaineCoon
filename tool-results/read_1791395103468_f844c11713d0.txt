   380→   380→function KittenCard({
   381→   381→  kitten,
   382→   382→  onSelect,
   383→   383→  onBook,
   384→   384→}: {
   385→   385→  kitten: Kitten;
   386→   386→  onSelect: () => void;
   387→   387→  onBook: () => void;
   388→   388→}) {
   389→   389→  const PersonalityIcon = PERSONALITY_ICON[kitten.personality];
   390→   390→  const gender = GENDER_LABEL[kitten.gender];
   391→   391→  const GenderIcon = gender.icon;
   392→   392→  const isAvailable = kitten.status === "available";
   393→   393→  const isExpected = kitten.status === "expected";
   394→   394→  const { isFavorite, toggle: toggleFav } = useFavorites();
   395→   395→  const { isComparing, toggle: toggleCompare, canAdd, max } = useCompare();
   396→   396→
   397→   397→  const fav = isFavorite(kitten.id);
   398→   398→  const comparing = isComparing(kitten.id);
   399→   399→
   400→   400→  const handleFav = (e: React.MouseEvent) => {
   401→   401→    e.stopPropagation();
   402→   402→    toggleFav(kitten.id);
   403→   403→    toast(fav ? "Убрано из избранного" : "Добавлено в избранное", {
   404→   404→      description: fav ? undefined : `${kitten.name} теперь в вашем списке.`,
   405→   405→    });
   406→   406→  };
   407→   407→
   408→   408→  const handleCompare = (e: React.MouseEvent) => {
   409→   409→    e.stopPropagation();
   410→   410→    if (!comparing && !canAdd) {
   411→   411→      toast.warning(`Можно сравнить максимум ${max} котят`, {
   412→   412→        description: "Уберите одного, чтобы добавить другого.",
   413→   413→      });
   414→   414→      return;
   415→   415→    }
   416→   416→    toggleCompare(kitten.id);
   417→   417→    toast(comparing ? "Убрано из сравнения" : "Добавлено к сравнению", {
   418→   418→      description: comparing ? undefined : `${kitten.name} в списке для сравнения.`,
   419→   419→    });
   420→   420→  };
   421→   421→
   422→   422→  return (
   423→   423→    <SpotlightCard className="rounded-2xl ring-1 ring-border h-full">
   424→   424→    <Card className="group overflow-hidden border-0 bg-card hover:shadow-lg transition-all duration-300 flex flex-col h-full">
   425→   425→      <div
   426→   426→        className="relative aspect-[4/3] overflow-hidden cursor-pointer"
   427→   427→        onClick={onSelect}
   428→   428→      >
   429→   429→        <Image
   430→   430→          src={kitten.imageUrl}
   431→   431→          alt={`${kitten.name} — ${kitten.colorLabel}`}
   432→   432→          fill
   433→   433→          sizes="(max-width: 640px) 100vw, (max-width: 1024px) 50vw, 33vw"
   434→   434→          className="object-cover transition-transform duration-500 group-hover:scale-105"
   435→   435→        />
   436→   436→        <div className="absolute inset-0 bg-gradient-to-t from-black/40 to-transparent" />
   437→   437→
   438→   438→        {/* Action buttons (top-right): favorite + compare */}
   439→   439→        <div className="absolute top-2.5 right-2.5 flex flex-col gap-1.5 z-10">
   440→   440→          <button
   441→   441→            type="button"
   442→   442→            onClick={handleFav}
   443→   443→            aria-label={fav ? "Убрать из избранного" : "Добавить в избранное"}
   444→   444→            aria-pressed={fav}
   445→   445→            className="flex h-8 w-8 items-center justify-center rounded-full bg-background/90 backdrop-blur-sm border border-border text-foreground hover:bg-background hover:scale-110 transition-all shadow-sm"
   446→   446→          >
   447→   447→            <Heart
   448→   448→              className={`h-4 w-4 transition-all ${
   449→   449→                fav ? "fill-accent text-accent scale-110" : "text-foreground"
   450→   450→              }`}
   451→   451→            />
   452→   452→          </button>
   453→   453→          <button
   454→   454→            type="button"
   455→   455→            onClick={handleCompare}
   456→   456→            aria-label={comparing ? "Убрать из сравнения" : "Добавить к сравнению"}
   457→   457→            aria-pressed={comparing}
   458→   458→            className={`flex h-8 w-8 items-center justify-center rounded-full backdrop-blur-sm border transition-all shadow-sm ${
   459→   459→              comparing
   460→   460→                ? "bg-secondary text-secondary-foreground border-secondary scale-110"
   461→   461→                : "bg-background/90 border-border text-foreground hover:bg-background hover:scale-110"
   462→   462→            }`}
   463→   463→          >
   464→   464→            <GitCompare className="h-4 w-4" />
   465→   465→          </button>
   466→   466→        </div>

   766→   766→function formatAge(iso: string): string {
   767→   767→  try {
   768→   768→    const birth = new Date(iso);
   769→   769→    const now = new Date();
   770→   770→    const months =
   771→   771→      (now.getFullYear() - birth.getFullYear()) * 12 +
   772→   772→      (now.getMonth() - birth.getMonth());
   773→   773→    if (months <= 0) return "новорождённый";
   774→   774→    if (months < 12) return `${months} мес.`;
   775→   775→    const years = Math.floor(months / 12);
   776→   776→    const remMonths = months % 12;
   777→   777→    return remMonths === 0
   778→   778→      ? `${years} ${years === 1 ? "год" : "года"}`
   779→   779→      : `${years} г. ${remMonths} мес.`;
   780→   780→  } catch {
   781→   781→    return "—";
   782→   782→  }
